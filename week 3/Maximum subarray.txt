class Solution {
public:
    int maxSubArray(vector<int>& nums) {
        int ans=nums[0];      
         int i,sum=0;
        for(i=0;i<nums.size();i++){
            sum=sum+nums[i];
            ans = max(ans, sum);
            if(sum<0){
                sum= 0;
            }
       

        }
        return ans;
       
    }
};
