#include<iostream>
#include<limits.h>
using namespace std;

void prefix(int a[],int n){
    int pre[10];
    int sum=0;
    int j;
    for(j=0;j<n;j++){
        sum += a[j];
        pre[j] = sum;
    }
    for(j = 0; j < n; j++)
    {
        cout << pre[j] << " ";
    }

}

int main(){
    int n,i,a[10] ;
    cout<< "Enter the size of input :";
    cin>> n;
    for(i=0;i<n;i++){
        cin>> a[i];
    }
    prefix(a,n);
}
