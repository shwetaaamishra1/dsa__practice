#include <iostream>
#include <algorithm>
using namespace std;

int maxdiff(int a[], int n)
{
    int ans = 0;
    int minElement = a[0];

    for(int j = 1; j < n; j++)
    {
        minElement = min(minElement, a[j]);

        ans = max(ans, a[j] - minElement);
    }

    return ans;
}

int main()
{
    int a[10], n;

    cout << "Enter the size of Array: ";
    cin >> n;

    for(int i = 0; i < n; i++)
    {
        cin >> a[i];
    }

    cout << "Maximum Difference = " << maxdiff(a, n);

    return 0;
}
