class Solution {
public:
    string sortSentence(string s) {

        // Maximum words according to the problem is 9
        vector<string> ans(10);

        string word = "";

        for (int i = 0; i < s.size(); i++) {

            // Build the word
            if (isalpha(s[i])) {
                word += s[i];
            }

            // Found the position digit
            else if (isdigit(s[i])) {
                int pos = s[i] - '0';
                ans[pos] = word;   // Store at its correct position
                word = "";         // Ready for the next word
            }
        }

        // Build the final answer
        string result = "";

        for (int i = 1; i <= 9; i++) {

            if (ans[i] != "") {

                if (result != "")
                    result += " ";

                result += ans[i];
            }
        }

        return result;
    }
};
