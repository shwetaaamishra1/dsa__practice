#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

vector<int> fact(int n)
{
    vector<int> ans(1, 1);

    while (n > 1)
    {
        int carry = 0;

        for (int i = 0; i < ans.size(); i++)
        {
            int res = ans[i] * n + carry;
            ans[i] = res % 10;
            carry = res / 10;
        }

        while (carry)
        {
            ans.push_back(carry % 10);
            carry /= 10;
        }

        n--;
    }

    reverse(ans.begin(), ans.end());
    return ans;
}

int main()
{
    int n;
    cout << "Enter the value: ";
    cin >> n;

    vector<int> ans = fact(n);

    for (int digit : ans)
        cout << digit;

    cout << endl;

    return 0;
}
